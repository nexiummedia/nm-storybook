Forms are the most critical interactive surface in web applications. Suboptimal form architectures introduce performance bottlenecks (such as re-rendering the entire page on every keystroke), inaccessible error messages, and fragile validation logic.

Nexium Media standardizes on **React Hook Form** paired with **Zod** schema validation for all client applications. This pairing provides uncontrolled, high-performance form inputs, strict TypeScript inference, and declarative schema rules.

## Technology Stack Rationale

| Tool | Role in Stack | Why Nexium Enforces It |
| :--- | :--- | :--- |
| **React Hook Form (RHF)** | Form state management | Uncontrolled form architecture minimizes component re-renders. Microscopic bundle size (~8.5kB). |
| **Zod** | Schema definition & validation | Single source of truth for both runtime validation and static TypeScript type inference. |
| `@hookform/resolvers` | Bridge library | Connects Zod schemas directly into RHF lifecycle with zero glue code. |

## Complete Production Example: Registration Form

Below is the standard pattern for a registration form with password confirmation, schema validation, accessible error announcements, and loading indicators:

```tsx
"use client";

import * as React from "react";
import { useForm } from "react-hook-form";
import { zodResolver } from "@hookform/resolvers/zod";
import { z } from "zod";
import { Button } from "@/components/ui/Button";

// 1. Declare Zod Validation Schema with refinement
export const RegisterSchema = z
  .object({
    email: z.string().min(1, "Email is required").email("Invalid email address"),
    password: z
      .string()
      .min(8, "Password must be at least 8 characters")
      .regex(/[A-Z]/, "Must contain at least one uppercase letter")
      .regex(/[0-9]/, "Must contain at least one numeric digit"),
    confirmPassword: z.string().min(1, "Please confirm your password"),
  })
  .refine((data) => data.password === data.confirmPassword, {
    message: "Passwords do not match",
    path: ["confirmPassword"], // Attaches error specifically to confirmPassword field
  });

export type RegisterFormData = z.infer<typeof RegisterSchema>;

export function RegisterForm() {
  const {
    register,
    handleSubmit,
    setError,
    formState: { errors, isSubmitting, isDirty, isValid },
  } = useForm<RegisterFormData>({
    resolver: zodResolver(RegisterSchema),
    mode: "onBlur", // Validates as users leave each input
    defaultValues: {
      email: "",
      password: "",
      confirmPassword: "",
    },
  });

  const onSubmit = async (data: RegisterFormData) => {
    try {
      const response = await fetch("/api/auth/register", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify(data),
      });

      if (!response.ok) {
        const errorPayload = await response.json();
        // Set server validation error on specific field
        if (errorPayload.field === "email") {
          setError("email", { type: "server", message: errorPayload.message });
          return;
        }
        throw new Error(errorPayload.message || "Registration failed");
      }

      window.location.href = "/dashboard";
    } catch (err: unknown) {
      setError("root", {
        type: "server",
        message: err instanceof Error ? err.message : "An unexpected server error occurred.",
      });
    }
  };

  return (
    <form onSubmit={handleSubmit(onSubmit)} noValidate className="max-w-md space-y-4">
      {errors.root && (
        <div role="alert" className="rounded-md bg-red-50 p-3 text-sm text-red-700">
          {errors.root.message}
        </div>
      )}

      {/* Email Field */}
      <div>
        <label htmlFor="reg-email" className="block text-sm font-medium text-slate-700">
          Corporate Email Address
        </label>
        <input
          {...register("email")}
          id="reg-email"
          type="email"
          autoComplete="email"
          aria-invalid={!!errors.email}
          aria-describedby={errors.email ? "email-error" : undefined}
          className="mt-1 block w-full rounded-md border border-slate-300 px-3 py-2 text-sm shadow-sm focus:border-blue-500 focus:outline-none focus:ring-1 focus:ring-blue-500"
        />
        {errors.email && (
          <p id="email-error" role="alert" className="mt-1 text-xs text-red-600">
            {errors.email.message}
          </p>
        )}
      </div>

      {/* Password Field */}
      <div>
        <label htmlFor="reg-password" className="block text-sm font-medium text-slate-700">
          Password
        </label>
        <input
          {...register("password")}
          id="reg-password"
          type="password"
          autoComplete="new-password"
          aria-invalid={!!errors.password}
          aria-describedby={errors.password ? "password-error" : undefined}
          className="mt-1 block w-full rounded-md border border-slate-300 px-3 py-2 text-sm shadow-sm focus:border-blue-500 focus:outline-none focus:ring-1 focus:ring-blue-500"
        />
        {errors.password && (
          <p id="password-error" role="alert" className="mt-1 text-xs text-red-600">
            {errors.password.message}
          </p>
        )}
      </div>

      {/* Confirm Password Field */}
      <div>
        <label htmlFor="reg-confirm" className="block text-sm font-medium text-slate-700">
          Confirm Password
        </label>
        <input
          {...register("confirmPassword")}
          id="reg-confirm"
          type="password"
          autoComplete="new-password"
          aria-invalid={!!errors.confirmPassword}
          aria-describedby={errors.confirmPassword ? "confirm-error" : undefined}
          className="mt-1 block w-full rounded-md border border-slate-300 px-3 py-2 text-sm shadow-sm focus:border-blue-500 focus:outline-none focus:ring-1 focus:ring-blue-500"
        />
        {errors.confirmPassword && (
          <p id="confirm-error" role="alert" className="mt-1 text-xs text-red-600">
            {errors.confirmPassword.message}
          </p>
        )}
      </div>

      <Button type="submit" isLoading={isSubmitting} disabled={!isDirty || isSubmitting} className="w-full">
        Create Enterprise Account
      </Button>
    </form>
  );
}
```

<Callout type="tip">
Always pass `noValidate` to `<form>` when using client-side schema validation. This disables default browser tooltips, ensuring consistent cross-browser error messages styled according to your design system.
</Callout>

## Controlled Components with `Controller`

Custom UI components (e.g., Radix Select, date pickers, rich-text inputs) do not expose native DOM refs. Use RHF's `<Controller>` component to bind them cleanly:

```tsx
import { Controller, useForm } from "react-hook-form";
import { Select } from "@/components/ui/Select";

export function TeamAssignmentForm() {
  const { control, handleSubmit } = useForm({ defaultValues: { role: "member" } });

  return (
    <form onSubmit={handleSubmit(console.log)}>
      <Controller
        name="role"
        control={control}
        render={({ field, fieldState: { error } }) => (
          <div>
            <label className="text-sm">Team Role</label>
            <Select
              value={field.value}
              onValueChange={field.onChange}
              onBlur={field.onBlur}
            >
              <Select.Item value="admin">Administrator</Select.Item>
              <Select.Item value="member">Team Member</Select.Item>
            </Select>
            {error && <p className="text-xs text-red-600">{error.message}</p>}
          </div>
        )}
      />
    </form>
  );
}
```

## Nested Forms with `FormProvider` & `useFormContext`

For complex, multi-section settings pages, avoid passing `register` and `errors` down through dozens of layers. Wrap the tree with `FormProvider`:

```tsx
import { useForm, FormProvider, useFormContext } from "react-hook-form";

// Deeply nested child field
function BillingAddressSubSection() {
  const { register, formState: { errors } } = useFormContext();
  return (
    <div>
      <input {...register("billingAddress.street")} placeholder="Street" />
      {errors.billingAddress?.street && <span>Invalid street</span>}
    </div>
  );
}

// Parent form
export function MasterAccountSettings() {
  const methods = useForm();
  return (
    <FormProvider {...methods}>
      <form onSubmit={methods.handleSubmit(console.log)}>
        <BillingAddressSubSection />
        <button type="submit">Save</button>
      </form>
    </FormProvider>
  );
}
```

## Dynamic Field Arrays with `useFieldArray`

For forms with dynamic line items (e.g., invoice line items, invite multiple team members):

```tsx
import { useForm, useFieldArray } from "react-hook-form";

interface FormValues {
  invites: { email: string }[];
}

export function MultiInviteForm() {
  const { register, control, handleSubmit } = useForm<FormValues>({
    defaultValues: { invites: [{ email: "" }] },
  });

  const { fields, append, remove } = useFieldArray({
    control,
    name: "invites",
  });

  return (
    <form onSubmit={handleSubmit(console.log)} className="space-y-3">
      {fields.map((field, index) => (
        <div key={field.id} className="flex gap-2">
          <input
            {...register(`invites.${index}.email` as const)}
            placeholder="colleague@nexiummedia.com"
            className="border p-2"
          />
          <button type="button" onClick={() => remove(index)}>Remove</button>
        </div>
      ))}
      <button type="button" onClick={() => append({ email: "" })}>Add Another</button>
      <button type="submit">Send Invites</button>
    </form>
  );
}
```

<Callout type="warning">
When rendering `fields.map()`, always use `key={field.id}` generated by `useFieldArray`. Never use the array index as the key; doing so will cause inputs to lose their values and focus states when items are removed or reordered.
</Callout>
