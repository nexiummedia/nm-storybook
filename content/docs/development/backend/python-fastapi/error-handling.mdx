Standardizing error responses in FastAPI ensures that frontend clients, mobile apps, and API consumers receive predictable error envelopes regardless of whether an error originates from schema validation, database integrity constraints, or unexpected server faults.

---

## 1. Domain Exception Hierarchy

Instead of scattering `HTTPException` instances across business services, define domain-specific exceptions:

```python title="app/utils/exceptions.py"
from typing import Any, Optional

class DomainException(Exception):
    """Base class for all application domain errors."""
    def __init__(
        self,
        message: str,
        code: str = "DOMAIN_ERROR",
        status_code: int = 400,
        details: Optional[Any] = None,
    ) -> None:
        super().__init__(message)
        self.message = message
        self.code = code
        self.status_code = status_code
        self.details = details

class EntityNotFoundException(DomainException):
    def __init__(self, entity_name: str, entity_id: Any) -> None:
        super().__init__(
            message=f"{entity_name} with identifier '{entity_id}' was not found.",
            code="ENTITY_NOT_FOUND",
            status_code=404,
        )

class DuplicateEntityException(DomainException):
    def __init__(self, message: str) -> None:
        super().__init__(
            message=message,
            code="DUPLICATE_ENTITY",
            status_code=409,
        )
```

## 2. Standard Error Response Schema

Every error returned by the API matches this standard envelope:

```python title="app/schemas/error.py"
from typing import Any, List, Optional
from pydantic import BaseModel

class ErrorDetail(BaseModel):
    field: Optional[str] = None
    message: str
    code: Optional[str] = None

class ErrorBody(BaseModel):
    code: str
    message: str
    details: Optional[List[ErrorDetail]] = None

class ErrorResponseEnvelope(BaseModel):
    error: ErrorBody
    meta: dict
```

## 3. Registering Global Exception Handlers

Register global handlers in `app/main.py` to intercept both framework-level and application-level errors:

```python title="app/main.py (Exception Handlers)"
from datetime import datetime, timezone
from fastapi import FastAPI, Request, status
from fastapi.exceptions import RequestValidationError
from fastapi.responses import JSONResponse
from sqlalchemy.exc import IntegrityError
import structlog
from app.utils.exceptions import DomainException
from app.config import settings

logger = structlog.get_logger()

def create_error_response(status_code: int, code: str, message: str, details=None, request_id: str = "") -> JSONResponse:
    return JSONResponse(
        status_code=status_code,
        content={
            "error": {
                "code": code,
                "message": message,
                "details": details or [],
            },
            "meta": {
                "requestId": request_id,
                "timestamp": datetime.now(timezone.utc).isoformat(),
            },
        },
    )

def setup_exception_handlers(app: FastAPI) -> None:
    # 1. Custom Domain Exceptions
    @app.exception_handler(DomainException)
    async def domain_exception_handler(request: Request, exc: DomainException) -> JSONResponse:
        request_id = getattr(request.state, "request_id", "")
        return create_error_response(
            status_code=exc.status_code,
            code=exc.code,
            message=exc.message,
            details=exc.details,
            request_id=request_id,
        )

    # 2. Pydantic Request Validation Errors (Format into standard envelope)
    @app.exception_handler(RequestValidationError)
    async def validation_exception_handler(request: Request, exc: RequestValidationError) -> JSONResponse:
        request_id = getattr(request.state, "request_id", "")
        formatted_details = []
        for error in exc.errors():
            loc = ".".join(str(x) for x in error.get("loc", []) if x != "body")
            formatted_details.append({
                "field": loc or "payload",
                "message": error.get("msg", "Invalid value"),
                "code": error.get("type", "value_error"),
            })

        return create_error_response(
            status_code=status.HTTP_422_UNPROCESSABLE_ENTITY,
            code="VALIDATION_FAILED",
            message="Request body or parameter validation failed",
            details=formatted_details,
            request_id=request_id,
        )

    # 3. SQLAlchemy Database Integrity Violations (Unique Constraint, Foreign Key)
    @app.exception_handler(IntegrityError)
    async def sqlalchemy_integrity_handler(request: Request, exc: IntegrityError) -> JSONResponse:
        request_id = getattr(request.state, "request_id", "")
        logger.error("Database integrity violation", exc_info=exc, request_id=request_id)
        return create_error_response(
            status_code=status.HTTP_409_CONFLICT,
            code="DATABASE_CONFLICT",
            message="Database constraint violation occurred",
            request_id=request_id,
        )

    # 4. Unhandled Catch-All Internal Server Errors (500)
    @app.exception_handler(Exception)
    async def generic_exception_handler(request: Request, exc: Exception) -> JSONResponse:
        request_id = getattr(request.state, "request_id", "")
        logger.error("Unhandled server exception", exc_info=exc, request_id=request_id)

        details = [{"stack": str(exc)}] if settings.ENVIRONMENT == "development" else []
        return create_error_response(
            status_code=status.HTTP_500_INTERNAL_SERVER_ERROR,
            code="INTERNAL_SERVER_ERROR",
            message="An unexpected server error occurred. Please contact support.",
            details=details,
            request_id=request_id,
        )
```

## 4. Structured Error Logging with `structlog`

All errors in production are emitted in structured JSON format with contextual request IDs:

```json
{
  "event": "Database integrity violation",
  "request_id": "req_8b91a27e",
  "level": "error",
  "timestamp": "2026-11-01T08:15:32.412Z",
  "exception": "psycopg2.errors.UniqueViolation: duplicate key value violates unique constraint 'uq_user_email'..."
}
```

<Callout type="error">
Never return raw database error messages or Python stack traces to API consumers in production. They expose internal database schemas, table names, and vulnerability targets to attackers.
</Callout>
